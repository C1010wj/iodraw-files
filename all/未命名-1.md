```mermaid
@startuml 数据库E-R图
!define Table(x) class x << (T,#FFAAAA) >>
!define PK(x) <u>x</u>
!define FK(x) #x#

skinparam class {
    BackgroundColor White
    ArrowColor #333
    BorderColor #333
}

Table(sys_user) {
    PK(user_id): BIGINT
    username: VARCHAR(50)
    password: VARCHAR(200)
    email: VARCHAR(100)
    create_time: DATETIME
    update_time: DATETIME
}

Table(category) {
    PK(category_id): BIGINT
    category_name: VARCHAR(50)
    type: TINYINT
    parent_id: BIGINT
    FK(user_id): BIGINT
    sort_order: INT
}

Table(income_expense) {
    PK(record_id): BIGINT
    FK(user_id): BIGINT
    FK(category_id): BIGINT
    amount: DECIMAL(12,2)
    type: TINYINT
    record_date: DATE
    remark: VARCHAR(200)
    create_time: DATETIME
    update_time: DATETIME
}

Table(budget) {
    PK(budget_id): BIGINT
    FK(user_id): BIGINT
    FK(category_id): BIGINT
    budget_amount: DECIMAL(12,2)
    budget_month: VARCHAR(7)
    create_time: DATETIME
}

sys_user "1" --> "0..*" income_expense : 拥有
sys_user "1" --> "0..*" category : 自定义
sys_user "1" --> "0..*" budget : 设置
category "1" --> "0..*" income_expense : 归类
category "1" --> "0..*" budget : 关联

note right of income_expense::amount
  支出存储为负数
  收入存储为正数
end note

note right of category::type
  0-支出分类
  1-收入分类
end note

@enduml
```
